-- Day 7 Assignment

create database if not exists day7_simple;
use day7_simple;

create table customers(
 customer_id int primary key,
 customer_name varchar(30)
);
create table hubs(
 hub_id int primary key,
 hub_name varchar(30)
);
create table carriers(
 carrier_id int primary key,
 carrier_name varchar(30)
);
create table employees(
 emp_id int primary key,
 emp_name varchar(30),
 manager_id int
);
create table shipments(
 shipment_id int primary key,
 customer_id int,
 carrier_id int,
 from_hub int,
 to_hub int,
 cost int
);
create table delivery_events(
 shipment_id int primary key,
 delay_days int
);

-- Task 1
select shipment_id,customer_name
from shipments inner join customers using(customer_id);

select shipment_id,carrier_name,delay_days
from shipments inner join carriers using(carrier_id)
inner join delivery_events using(shipment_id)
where delay_days>2;

select hub_name from hubs left join shipments on hub_id=from_hub
where shipment_id is null;

select customer_name,shipment_id
from shipments right join customers using(customer_id);

select e.emp_name,m.emp_name as manager
from employees e left join employees m on e.manager_id=m.emp_id;

select shipment_id,hub_name
from shipments inner join hubs on from_hub=hub_id;

select shipment_id,hub_name
from shipments inner join hubs on to_hub=hub_id;

select carrier_name,shipment_id
from carriers left join shipments using(carrier_id);

select shipment_id,customer_name,cost
from shipments inner join customers using(customer_id)
where cost>150;

select shipment_id,delay_days
from shipments inner join delivery_events using(shipment_id)
where delay_days=0;

-- Task 2
select customer_name from customers c where not exists(
 select * from hubs h where not exists(
  select * from shipments
  where customer_id=c.customer_id and to_hub=h.hub_id
 )
);

select customer_name from customers c where exists(
 select * from shipments where customer_id=c.customer_id
);

select carrier_name from carriers where carrier_id in(
 select carrier_id from shipments inner join delivery_events using(shipment_id)
 group by carrier_id
 having avg(delay_days)>(select avg(delay_days) from delivery_events)
);

select shipment_id,cost from shipments
where cost>any(select cost from shipments where customer_id=1);

select shipment_id,cost from shipments
where cost>all(select cost from shipments where customer_id=1);

-- Task 3
explain analyze
select customer_id,customer_name from customers where customer_id in(
 select customer_id from shipments where cost>150
);

explain analyze
select customer_id,customer_name
from customers inner join shipments using(customer_id)
where cost>150 group by customer_id,customer_name;

explain analyze
select carrier_id,carrier_name from carriers where carrier_id in(
 select carrier_id from shipments where cost>150
);

explain analyze
select carrier_id,carrier_name
from carriers inner join shipments using(carrier_id)
where cost>150 group by carrier_id,carrier_name;

-- Task 4
select hub_name,(
 select max(cost) from shipments
 where to_hub=h.hub_id and cost<(
  select max(cost) from shipments where to_hub=h.hub_id
 )
) as second_highest_cost from hubs h;
